import { NextResponse } from "next/server";
import { deleteProduct, getProduct, updateProduct } from "@/lib/products";
import { validateProduct } from "@/lib/validation";
import { revalidateTag } from "next/cache";

type Ctx = { params: Promise<{ id: string }> };

export async function GET(_req: Request, { params }: Ctx) {
  const { id } = await params;
  try {
    const product = await getProduct(id);
    if (!product) return NextResponse.json({ error: "Introuvable" }, { status: 404 });
    return NextResponse.json(product);
  } catch {
    return NextResponse.json({ error: "Erreur serveur" }, { status: 500 });
  }
}

export async function PUT(request: Request, { params }: Ctx) {
  const { id } = await params;

  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ errors: ["JSON invalide"] }, { status: 400 });
  }

  const result = validateProduct(body);
  if (!result.ok) {
    return NextResponse.json({ errors: result.errors }, { status: 400 });
  }

  try {
    const updated = await updateProduct(id, result.data);
    if (!updated) return NextResponse.json({ error: "Introuvable" }, { status: 404 });
    revalidateTag("products", { expire: 0 });
    return NextResponse.json(updated);
  } catch {
    return NextResponse.json({ error: "Erreur serveur" }, { status: 500 });
  }
}

export async function DELETE(_req: Request, { params }: Ctx) {
  const { id } = await params;
  try {
    const ok = await deleteProduct(id);
    if (!ok) return NextResponse.json({ error: "Introuvable" }, { status: 404 });
    revalidateTag("products", { expire: 0 });
    return new Response(null, { status: 204 });
  } catch {
    return NextResponse.json({ error: "Erreur serveur" }, { status: 500 });
  }
}

import { revalidateTag } from "next/cache";

  try {
    return NextResponse.json(await listProducts());

    return NextResponse.json({ error: "Erreur serveur" }, { status: 500 });

}

export async function POST(request: Request) {
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ errors: ["JSON invalide"] }, { status: 400 });
  }

  const result = validateProduct(body);
  if (!result.ok) {
    return NextResponse.json({ errors: result.errors }, { status: 400 });
  }

  try {
    const created = await createProduct(result.data);
    revalidateTag("products", { expire: 0 });
    return NextResponse.json(created, { status: 201 });
  } catch {
    return NextResponse.json({ error: "Erreur serveur" }, { status: 500 });
  }
}

"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import type { Product } from "@/lib/types";

export default function ProductForm({ product }: { product?: Product }) {
  const router = useRouter();
  const [errors, setErrors] = useState<string[]>([]);
  const [loading, setLoading] = useState(false);

  async function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setLoading(true);
    setErrors([]);

    const f = new FormData(e.currentTarget);
    const body = {
      brand: String(f.get("brand")),
      model: String(f.get("model")),
      price: Number(f.get("price")),
      stock: Number(f.get("stock")),
      description: String(f.get("description")),
    };

    const res = await fetch(product ? `/api/products/${product.id}` : "/api/products", {
      method: product ? "PUT" : "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });

    if (!res.ok) {
      const data = await res.json().catch(() => ({}));
      setErrors(data.errors ?? [data.error ?? "Erreur"]);
      setLoading(false);
      return;
    }

    const saved: Product = await res.json();
    router.push(`/products/${saved.id}`);
    router.refresh();
  }

  return (
    <form onSubmit={onSubmit} className="card">
      <label>Marque <input name="brand" defaultValue={product?.brand} required /></label>
      <label>Modèle <input name="model" defaultValue={product?.model} required /></label>
      <label>Prix (€) <input name="price" type="number" step="0.01" min="0" defaultValue={product?.price} required /></label>
      <label>Stock <input name="stock" type="number" step="1" min="0" defaultValue={product?.stock ?? 0} required /></label>
      <label>Description <textarea name="description" defaultValue={product?.description} /></label>
      {errors.map((err) => <p key={err} style={{ color: "crimson" }}>{err}</p>)}
      <button type="submit" disabled={loading}>{product ? "Enregistrer" : "Ajouter"}</button>
    </form>
  );
}

import { ClientResponseError, type RecordModel } from "pocketbase";
import { getPb } from "./pocketbase";
import type { Product, ProductInput } from "./types";

function toProduct(r: RecordModel): Product {
  return {
    id: r.id,
    brand: r.brand,
    model: r.model,
    price: r.price,
    description: r.description ?? "",
    stock: r.stock ?? 0,
  };
}

function isNotFound(e: unknown) {
  return e instanceof ClientResponseError && e.status === 404;
}

export async function listProducts(): Promise<Product[]> {
  const base = process.env.PB_URL ?? "http://127.0.0.1:8090";
  const res = await fetch(
    `${base}/api/collections/laptops/records?perPage=500&sort=brand,model`,
    { next: { tags: ["products"], revalidate: 300 } }
  );
  if (!res.ok) throw new Error("PocketBase error");
  const data = await res.json();
  return (data.items as RecordModel[]).map(toProduct);
}

export async function getProduct(id: string): Promise<Product | null> {
  try {
    return toProduct(await getPb().collection("laptops").getOne(id));
  } catch (e) {
    if (isNotFound(e)) return null;
    throw e;
  }
}

export async function createProduct(data: ProductInput): Promise<Product> {
  return toProduct(await getPb().collection("laptops").create(data));
}

export async function updateProduct(id: string, data: ProductInput): Promise<Product | null> {
  try {
    return toProduct(await getPb().collection("laptops").update(id, data));
  } catch (e) {
    if (isNotFound(e)) return null;
    throw e;
  }
}

export async function deleteProduct(id: string): Promise<boolean> {
  try {
    await getPb().collection("laptops").delete(id);
    return true;
  } catch (e) {
    if (isNotFound(e)) return false;
    throw e;
  }
}

import { createClient } from "redis";

function makeClient() {
  return createClient({ url: process.env.REDIS_URL });
}

const globalForRedis = globalThis as unknown as {
  redisClient?: ReturnType<typeof makeClient>;
};

export async function getRedis() {
  if (!globalForRedis.redisClient) {
    const client = makeClient();
    client.on("error", (err) => console.error("Redis error", err));
    await client.connect();
    globalForRedis.redisClient = client;
  }
  return globalForRedis.redisClient;
}
===== lib/stats.ts
import type { Product } from "./types";

export function stockByBrand(products: Product[]) {
  const totals: Record<string, number> = {};
  for (const p of products) {
    totals[p.brand] = (totals[p.brand] ?? 0) + p.stock;
  }
  return Object.entries(totals).map(([brand, stock]) => ({ brand, stock }));
}

export function pricePerModel(products: Product[]) {
  return products
    .map((p) => ({ label: `${p.brand} ${p.model}`, price: p.price }))
    .sort((a, b) => a.price - b.price);
}

===== lib/validation.ts
import type { ProductInput } from "./types";

export type ValidationResult =
  | { ok: true; data: ProductInput }
  | { ok: false; errors: string[] };

export function validateProduct(body: unknown): ValidationResult {
  const errors: string[] = [];

  if (typeof body !== "object" || body === null || Array.isArray(body)) {
    return { ok: false, errors: ["Le corps doit être un objet JSON"] };
  }
  const b = body as Record<string, unknown>;

  const brand = typeof b.brand === "string" ? b.brand.trim() : "";
  if (brand.length < 1 || brand.length > 50) {
    errors.push("brand : texte de 1 à 50 caractères");
  }

  const model = typeof b.model === "string" ? b.model.trim() : "";
  if (model.length < 1 || model.length > 100) {
    errors.push("model : texte de 1 à 100 caractères");
  }

  const price = b.price;
  if (typeof price !== "number" || !Number.isFinite(price) || price < 0 || price > 100000) {
    errors.push("price : nombre entre 0 et 100000");
  }

  const description = b.description === undefined ? "" : b.description;
  if (typeof description !== "string" || description.length > 1000) {
    errors.push("description : texte de 1000 caractères maximum");
  }

  const stock = b.stock === undefined ? 0 : b.stock;
  if (typeof stock !== "number" || !Number.isInteger(stock) || stock < 0 || stock > 100000) {
    errors.push("stock : entier entre 0 et 100000");
  }

  if (errors.length > 0) return { ok: false, errors };

  return {
    ok: true,
    data: {
      brand,
      model,
      price: price as number,
      description: (description as string).trim(),
      stock: stock as number,
    },
  };
}

===== tests/TEST_PLAN.md
# Test Plan

Tools: Vitest + React Testing Library (units, API, components, pages), Playwright (end-to-end in a real browser), coverage with v8.
The tests were defined with the help of an AI (see tests/AI_PROMPTS.md).

## What we test and why
- lib/validation.ts: this is the most critical business rule, everyone goes through it
- lib/products.ts: PocketBase access, case 404, and errors
- lib/stats.ts: graph calculations
- lib/redis.ts and lib/homeContent.ts: home page cache and cases where Redis is broken
- app/api/** : HTTP codes (200, 201, 204, 400, 404, 500) and cache invalidation
- components/** : form, deletion, charts
- app/**/page.tsx: rendering pages with simulated data
- Playwright: real paths (create, edit, delete, admin, 404, API)

## Various test chains (validation and form)
blank, spaces only, 1 character, 50 / 51 characters, 100 / 101, 1000 / 1001,
numbers alone ("12345", "0"), wildcards and symbols (% * _ ?), SQL injection ("' OR 1=1 --"),
HTML (<script>alert(1)</script>), paths ("../../etc/passwd"), templates ("{{7*7}}"),
accents and unicode, Japanese, emojis, line breaks, tabs, backslash.

import { describe, expect, it } from "vitest";
import { validateProduct } from "@/lib/validation";

const base = { brand: "Dell", model: "XPS 13", price: 1199, description: "Bon laptop", stock: 8 };
const check = (overrides: Record<string, unknown>) => validateProduct({ ...base, ...overrides });
const errorsOf = (r: ReturnType<typeof validateProduct>) => (r.ok ? [] : r.errors);

const weirdStrings = [
  "%", "*", "_", "?", "0", "12345", "' OR 1=1 --", "<script>alert(1)</script>",
  "Ünïcödé é à ç", "日本語", "😀💻", "a\nb", "\ttabs", "../../etc/passwd", "{{7*7}}", "\\",
];

describe("validateProduct : cas valides", () => {
  it("V1 accepte un produit complet", () => {
    expect(check({})).toEqual({ ok: true, data: base });
  });

  it("V2 nettoie les espaces autour des textes", () => {
    const r = check({ brand: "  Dell  ", model: " XPS ", description: "  ok  " });
    expect(r).toEqual({ ok: true, data: { ...base, brand: "Dell", model: "XPS", description: "ok" } });
  });

  it("V3 description absente donne vide et stock absent donne 0", () => {
    const r = validateProduct({ brand: "A", model: "B", price: 1 });
    expect(r).toEqual({ ok: true, data: { brand: "A", model: "B", price: 1, description: "", stock: 0 } });
  });

  it("V4 accepte les limites exactes", () => {
    const r = check({
      brand: "a".repeat(50), model: "b".repeat(100), description: "c".repeat(1000),
      price: 100000, stock: 100000,
    });
    expect(r.ok).toBe(true);
    expect(check({ price: 0, stock: 0 }).ok).toBe(true);
  });
});

describe("validateProduct : chaînes étranges", () => {
  it.each(weirdStrings)("V5 accepte %j comme marque et modèle", (s) => {
    const r = check({ brand: s, model: s });
    expect(r.ok).toBe(true);
    if (r.ok) {
      expect(r.data.brand).toBe(s.trim());
      expect(r.data.model).toBe(s.trim());
    }
  });
});

describe("validateProduct : cas invalides", () => {
  const cases = [
    { label: "brand vide", o: { brand: "" }, field: "brand" },
    { label: "brand espaces seuls", o: { brand: "   " }, field: "brand" },
    { label: "brand 51 caractères", o: { brand: "a".repeat(51) }, field: "brand" },
    { label: "brand nombre", o: { brand: 123 }, field: "brand" },
    { label: "brand null", o: { brand: null }, field: "brand" },
    { label: "model vide", o: { model: "" }, field: "model" },
    { label: "model 101 caractères", o: { model: "b".repeat(101) }, field: "model" },
    { label: "model nombre", o: { model: 42 }, field: "model" },
    { label: "price négatif", o: { price: -1 }, field: "price" },
    { label: "price NaN", o: { price: NaN }, field: "price" },
    { label: "price Infinity", o: { price: Infinity }, field: "price" },
    { label: "price texte", o: { price: "100" }, field: "price" },
    { label: "price trop grand", o: { price: 100001 }, field: "price" },
